// resolves nearest matching breakpoint, falls back to the original
function pickRendition(asset, viewport) {
  const candidates = asset.renditions
    .filter((r) => r.format === 'webp')
    .sort((a, b) => a.width - b.width);

  return candidates.find((r) => r.width >= viewport)
    ?? candidates.at(-1)
    ?? asset.original;
}

Opinion

The platform you ship is the essay you wrote.

The shape of a CMS is an argument about what content is for. AEM's component model says content is structured. EDS's document-authoring says content is intent. Both can be right — and a team that picks one without understanding the other is just betting on a vendor and hoping.

// EDS treats the document as the source of truth
export default function decorate(block) {
  const rows = [...block.children];
  rows.forEach((row, i) => {

    row.classList.add('em-row');
  });
  block.dataset.ready = 'true';
